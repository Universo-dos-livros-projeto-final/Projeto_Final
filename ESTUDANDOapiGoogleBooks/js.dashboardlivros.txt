// 6. src/routes/books.js
import { PrismaClient } from '@prisma/client';
import { fetchBookDetails } from '../services/googleBooksService.js';

const prisma = new PrismaClient();

export default async function (fastify, options) {
  // Listar todos os livros
  fastify.get('/', async (request, reply) => {
    try {
      const books = await prisma.book.findMany({
        orderBy: {
          createdAt: 'desc'
        }
      });
      
      return books;
    } catch (error) {
      fastify.log.error(error);
      return reply.code(500).send({ error: 'Erro ao buscar livros' });
    }
  });
  
  // Buscar livro por ID
  fastify.get('/:id', async (request, reply) => {
    const { id } = request.params;
    
    try {
      const book = await prisma.book.findUnique({
        where: { id }
      });
      
      if (!book) {
        return reply.code(404).send({ error: 'Livro não encontrado' });
      }
      
      return book;
    } catch (error) {
      fastify.log.error(error);
      return reply.code(500).send({ error: 'Erro ao buscar livro' });
    }
  });
  
  // Adicionar novo livro
  fastify.post('/', async (request, reply) => {
    const bookData = request.body;
    
    try {
      // Verificar se o livro já existe
      if (bookData.googleBooksId) {
        const existingBook = await prisma.book.findUnique({
          where: { googleBooksId: bookData.googleBooksId }
        });
        
        if (existingBook) {
          return reply.code(409).send({ 
            error: 'Este livro já está na sua coleção',
            book: existingBook 
          });
        }
      }
      
      // Se tiver apenas o ID do Google Books, buscar detalhes
      if (bookData.googleBooksId && !bookData.title) {
        const bookDetails = await fetchBookDetails(bookData.googleBooksId);
        Object.assign(bookData, bookDetails);
      }
      
      const newBook = await prisma.book.create({
        data: bookData
      });
      
      return reply.code(201).send(newBook);
    } catch (error) {
      fastify.log.error(error);
      return reply.code(500).send({ error: 'Erro ao adicionar livro' });
    }
  });
  
  // Atualizar livro
  fastify.put('/:id', async (request, reply) => {
    const { id } = request.params;
    const bookData = request.body;
    
    try {
      const updatedBook = await prisma.book.update({
        where: { id },
        data: bookData
      });
      
      return updatedBook;
    } catch (error) {
      fastify.log.error(error);
      
      if (error.code === 'P2025') {
        return reply.code(404).send({ error: 'Livro não encontrado' });
      }
      
      return reply.code(500).send({ error: 'Erro ao atualizar livro' });
    }
  });
  
  // Remover livro
  fastify.delete('/:id', async (request, reply) => {
    const { id } = request.params;
    
    try {
      await prisma.book.delete({
        where: { id }
      });
      
      return reply.code(204).send();
    } catch (error) {
      fastify.log.error(error);
      
      if (error.code === 'P2025') {
        return reply.code(404).send({ error: 'Livro não encontrado' });
      }
      
      return reply.code(500).send({ error: 'Erro ao remover livro' });
    }
  });
}

// 7. src/services/googleBooksService.js
import fetch from 'node-fetch';

const GOOGLE_BOOKS_API = 'https://www.googleapis.com/books/v1/volumes';
const API_KEY = process.env.GOOGLE_BOOKS_API_KEY;

export async function fetchBookDetails(googleBooksId) {
  try {
    const url = `${GOOGLE_BOOKS_API}/${googleBooksId}${API_KEY ? `?key=${API_KEY}` : ''}`;
    const response = await fetch(url);
    
    if (!response.ok) {
      throw new Error(`Erro na API do Google Books: ${response.status}`);
    }
    
    const book = await response.json();
    const volumeInfo = book.volumeInfo;
    
    return {
      title: volumeInfo.title,
      authors: volumeInfo.authors || [],
      description: volumeInfo.description || '',
      thumbnail: volumeInfo.imageLinks?.thumbnail || '',
      publishedDate: volumeInfo.publishedDate || '',
      pageCount: volumeInfo.pageCount || 0,
      categories: volumeInfo.categories || []
    };
  } catch (error) {
    console.error('Erro ao buscar detalhes do livro:', error);
    throw error;
  }
}
